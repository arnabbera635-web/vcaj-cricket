rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function admin() {
      return signedIn()
        && request.auth.token.email == 'vcajofficial@gmail.com';
    }

    function member() {
      return signedIn()
        && exists(
          /databases/$(database)/documents/memberAccounts/$(request.auth.uid)
        );
    }

    // ---------- Per-person permissions (all flags live in memberAccounts/{uid} and only Admin can write them) ----------

    function acct() {
      return get(
        /databases/$(database)/documents/memberAccounts/$(request.auth.uid)
      ).data;
    }

    function allowed(perm) {
      return member()
        && acct().get('mustChangePassword', true) == false
        && acct().get('accessOn', false) == true
        && acct().get(perm, false) == true;
    }

    function staff() {
      return admin() || allowed('permView');
    }

    // a non-admin write is accepted only together with its activityLog entry in the same batch
    function logged() {
      return existsAfter(
          /databases/$(database)/documents/activityLog/$(request.resource.data.logId)
        )
        && getAfter(
          /databases/$(database)/documents/activityLog/$(request.resource.data.logId)
        ).data.uid == request.auth.uid;
    }

    match /liveMatches/{id} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /liveMatches/{id}/events/{eventId} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /matches/{id} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /matches/{id}/events/{eventId} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /players/{id} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /awards/{id} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /teams/{id} {
      allow read: if true;
      allow create, update, delete: if admin();
    }

    match /seasons/{id} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /tournaments/{id} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /publicMatchRecords/{id} {
      allow read: if true;
      allow create, update: if admin();
      allow delete: if false;
    }

    match /teamRegistrations/{id} {
      allow create: if
        request.resource.data.season == '2026'
        && request.resource.data.status == 'pending'
        && request.resource.data.receiptType == 'OFFICIAL_PAYMENT_RECEIPT'
        && request.resource.data.uniqueId is string
        && request.resource.data.verificationCode is string
        && request.resource.data.teamName is string
        && request.resource.data.whatsapp is string
        && request.resource.data.location is string

        && request.resource.data.entryFee.total == 3501
        && request.resource.data.entryFee.paid is number
        && request.resource.data.entryFee.due is number
        && request.resource.data.entryFee.paid >= 2000
        && request.resource.data.entryFee.paid <= 3501
        && request.resource.data.entryFee.due ==
           3501 - request.resource.data.entryFee.paid

        && request.resource.data.kasanmani.total == 1500
        && request.resource.data.kasanmani.paid is number
        && request.resource.data.kasanmani.due is number
        && request.resource.data.kasanmani.paid >= 0
        && request.resource.data.kasanmani.paid <= 1500
        && request.resource.data.kasanmani.due ==
           1500 - request.resource.data.kasanmani.paid

        && request.resource.data.totalPaid is number
        && request.resource.data.totalDue is number
        && request.resource.data.totalPaid ==
           request.resource.data.entryFee.paid
           + request.resource.data.kasanmani.paid
        && request.resource.data.totalDue ==
           request.resource.data.entryFee.due
           + request.resource.data.kasanmani.due

        && request.resource.data.paymentHistory is list;

      // office bearers may view registration payments; only admin may change them
      allow read: if staff();
      allow update, delete: if admin();
    }

    match /receiptVerifications/{code} {
      allow read: if true;

      allow create: if
        request.resource.data.verificationCode == code
        && code.matches('RC-[A-F0-9]{32}')
        && request.resource.data.uniqueId is string
        && request.resource.data.teamName is string

        && request.resource.data.entryPaid is number
        && request.resource.data.entryPaid >= 2000
        && request.resource.data.entryPaid <= 3501
        && request.resource.data.entryDue is number
        && request.resource.data.entryDue ==
           3501 - request.resource.data.entryPaid

        && request.resource.data.kasanPaid is number
        && request.resource.data.kasanPaid >= 0
        && request.resource.data.kasanPaid <= 1500
        && request.resource.data.kasanDue is number
        && request.resource.data.kasanDue ==
           1500 - request.resource.data.kasanPaid

        && request.resource.data.totalPaid is number
        && request.resource.data.totalPaid ==
           request.resource.data.entryPaid
           + request.resource.data.kasanPaid

        && request.resource.data.totalDue is number
        && request.resource.data.totalDue ==
           request.resource.data.entryDue
           + request.resource.data.kasanDue

        && request.resource.data.status == 'pending'
        && request.resource.data.season == '2026'
        && request.resource.data.receiptType == 'OFFICIAL_PAYMENT_RECEIPT';

      allow update, delete: if admin();
    }

    match /members/{id} {
      allow read: if staff()
        || (
          member()
          && get(
            /databases/$(database)/documents/memberAccounts/$(request.auth.uid)
          ).data.memberId == id
        );
      allow create, delete: if admin();
      // a person allowed to issue receipts may only mark the fee status
      allow update: if admin()
        || (
          (allowed('permReceiptNew') || allowed('permReceiptEdit'))
          && request.resource.data.diff(resource.data)
             .affectedKeys()
             .hasOnly(['feeStatus'])
        );
    }

    match /memberDirectory/{id} {
      allow read: if member() || admin();
      allow create, delete: if admin();
      allow update: if admin()
        || (
          (allowed('permReceiptNew') || allowed('permReceiptEdit'))
          && request.resource.data.diff(resource.data)
             .affectedKeys()
             .hasOnly(['paymentStatus'])
        );
    }

    match /tournamentAccounts/{id} {
      allow read: if admin() || member();
      allow create, update: if admin();
      allow delete: if admin();
    }

    match /memberAccounts/{uid} {
      allow read: if admin()
        || (
          signedIn()
          && request.auth.uid == uid
        );

      allow create, delete: if admin();

      allow update: if admin()
        || (
          signedIn()
          && request.auth.uid == uid
          && request.resource.data.diff(resource.data)
             .affectedKeys()
             .hasOnly([
               'mustChangePassword',
               'passwordChangedAt',
               'updatedAt'
             ])
          && request.resource.data.mustChangePassword == false
        );
    }

    // ---------- Receipts, vouchers and the activity log ----------

    match /paymentReceipts/{id} {
      allow read: if staff();
      allow create: if admin()
        || (
          allowed('permReceiptNew')
          && logged()
          && request.resource.data.createdByUid == request.auth.uid
        );
      allow update: if admin()
        || (
          allowed('permReceiptEdit')
          && logged()
          && request.resource.data.editedByUid == request.auth.uid
        );
      // only receipts the Admin has marked as trial can ever be deleted
      allow delete: if admin() && resource.data.get('trial', false) == true;
    }

    match /receiptChecks/{id} {
      allow get: if true;
      allow list: if false;
      allow create, update: if admin()
        || allowed('permReceiptNew')
        || allowed('permReceiptEdit');
      allow delete: if false;
    }

    match /paymentVouchers/{id} {
      allow read: if staff();
      allow create: if admin()
        || (
          allowed('permVoucherNew')
          && logged()
          && request.resource.data.createdByUid == request.auth.uid
          && request.resource.data.status == 'pending'
        );
      allow update: if admin()
        || (
          allowed('permVoucherEdit')
          && logged()
          && request.resource.data.editedByUid == request.auth.uid
          && resource.data.get('status', 'approved') in ['pending', 'rejected']
          && request.resource.data.status == 'pending'
        )
        || (
          allowed('permVoucherApprove')
          && logged()
          && resource.data.get('status', 'approved') == 'pending'
          && request.resource.data.status in ['approved', 'rejected']
          && request.resource.data.reviewedByUid == request.auth.uid
          && request.resource.data.diff(resource.data)
             .affectedKeys()
             .hasOnly([
               'status',
               'reviewedByUid',
               'reviewedByName',
               'reviewedByOffice',
               'reviewedAtMs',
               'reviewNote',
               'logId'
             ])
        );
      allow delete: if false;
    }

    match /activityLog/{id} {
      allow read: if admin();
      allow create: if request.resource.data.uid == request.auth.uid
        && request.resource.data.at == request.time
        && (
          (admin() && request.resource.data.office == 'admin')
          || (
            member()
            && acct().get('accessOn', false) == true
            && request.resource.data.office == acct().get('office', '')
          )
        );
      allow update, delete: if false;
    }

    // ---------- Optional pages ----------

    match /sponsors/{id} {
      allow read: if true;
      allow create, update, delete: if admin();
    }

    match /wagonShots/{id} {
      allow read: if true;
      allow create, delete: if admin();
      allow update: if false;
    }

    match /notices/{id} {
      allow read: if true;
      allow create, update, delete: if admin();
    }

    match /broadcasts/{id} {
      allow read, create, update: if admin();
      allow delete: if false;
    }

    match /letters/{id} {
      allow read, create, update: if admin();
      allow delete: if false;
    }

    match /letterSettings/{id} {
      allow read, create, update: if admin();
      allow delete: if false;
    }

    match /prizeList/{id} {
      allow read: if true;
      allow create, update, delete: if admin();
    }

    match /gallery/{id} {
      allow read: if true;
      allow create, update, delete: if admin();
    }
  }
}
